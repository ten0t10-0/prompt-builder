LIST --i Colors
	ENTRY --p Red
	ENTRY --p Blue
	ENTRY --p Green
	ENTRY --p Black
	ENTRY --p White
END

TAB --i Meta
	SELECT --i Artists --prefix by
		SINGLE --i Some artist --p some artist
	END

	SELECT --i Theme --sort 0
		SINGLE --i Medieval --p medieval --a 1
		SINGLE --i Fantasy --p fantasy --a 1
		SINGLE --i Cyberpunk --p cyberpunk
	END

	ROW
		COLUMN --scale 2
			EDIT --i Style --pa flat color --pb anime coloring --r 0.7
		END
		COLUMN --scale 3
			EDIT_LINK --i Coloring (Linked: Style) --link Style --pa muted color --pb colorful
		END
	END
END

TAB --i People
	SINGLE --i Single prompt --a 1 --p 1boy --s 1
	SEPARATOR
	EDIT --i Skin shade --pa dark --pb pale --r 0.4 --postfix skin
	SEPARATOR
	DUAL --i Hair style --pp low ponytail --pn short hair --sn 0.8
	SEPARATOR
	SELECT --i Clothing
		FROM_LIST --i Colors --postfix clothes
		SINGLE --i Formal --p formal clothes
		SINGLE --i Overalls --p overalls
	END
END
